import re
import json

with open("raw.txt", "r", encoding="utf-8") as f:
    text = f.read()


prices = re.findall(r"\d[\d\s]*,\d{2}", text)


price_values = [float(p.replace(" ", "").replace(",", ".")) for p in prices]


products = re.findall(r"\d+\.\n(.+)", text)


total = sum(price_values)


datetime = re.search(r"\d{2}\.\d{2}\.\d{4}\s\d{2}:\d{2}:\d{2}", text)


payment = re.search(r"(Банковская карта|Наличные)", text)


data = {
    "products": products,
    "prices": price_values,
    "total_calculated": total,
    "date_time": datetime.group() if datetime else None,
    "payment_method": payment.group() if payment else None
}

print(json.dumps(data, indent=4, ensure_ascii=False))
